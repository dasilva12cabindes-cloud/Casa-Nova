async function finalizar(){ const sessao = await db.auth.getSession(); if(!sessao.data.session){ alert("Faça login primeiro."); location.href = "login.html"; return; } const carrinho = lerCarrinho(); if(!carrinho.length){ alert("Carrinho vazio."); return; } let total = 0; carrinho.forEach(function(i){ total += i.preco * i.qtd; }); const pedido = await db.from("pedidos").insert({ user_id: sessao.data.session.user.id, nome_cliente: sessao.data.session.user.email, telefone: "", tipo_entrega: "Loja", total: total }).select().single(); if(pedido.error){ alert("Erro: " + pedido.error.message); return; } const itens = carrinho.map(function(item){ return { pedido_id: pedido.data.id, produto_id: item.id, nome_produto: item.nome, preco_unitario: item.preco, quantidade: item.qtd }; }); const salvarItens = await db.from("itens_pedido").insert(itens); if(salvarItens.error){ alert("Erro itens: " + salvarItens.error.message); return; } localStorage.removeItem("carrinho"); alert("Pedido realizado com sucesso! Nº " + pedido.data.id); location.href = "index.html"; }
